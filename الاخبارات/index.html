  القسم: "كمي - مقارنات هندسية وجبرية",
  نص_السؤال: "قارن بين: القيمة الأولى (س + ص)² | القيمة الثانية (س² + ص²)",
  المعطيات: "علماً بأن: س > 0 و ص > 0 (أعداد موجبة تماماً)",

    return {
      الإجابة_الصحيحة: "(أ) القيمة الأولى أكبر من القيمة الثانية",
      زمن_الحل_المثالي: "12 ثانية",
      نسبة_التقفيل: 100%
    };
  }
});
